React Native 실시간 소셜 모듈

Realtime Thread Integration

여행별 단체 메신저와 개별 일정 댓글 시스템을 설계할 때 Firebase Firestore의 실시간 스트리밍 리스너(`onSnapshot`)를 어떻게 바인딩하는지 명세화한 Native 뼈대 구조입니다.

// ActiveTripChatScreen.tsx - 실시간 단톡방 컴포넌트 
{`import React, { useEffect, useState } from 'react';
import { View, Text, TextInput, FlatList, Image } from 'react-native';
import { db } from './firebaseConfig';
import { collection, addDoc, query, orderBy, onSnapshot } from 'firebase/firestore';

export default function TripChat({ tripId, currentUser }) {
  const [messages, setMessages] = useState([]);
  const [text, setText] = useState('');

  useEffect(() => {
    const q = query(
      collection(db, 'trips', tripId, 'chats'),
      orderBy('createdAt', 'asc')
    );
    const unsubscribe = onSnapshot(q, (snapshot) => {
      const msgs = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
      setMessages(msgs);
    });
    return unsubscribe;
  }, [tripId]);

  const sendMessage = async () => {
    if (!text.trim()) return;
    await addDoc(collection(db, 'trips', tripId, 'chats'), {
      text,
      senderId: currentUser.id,
      senderName: currentUser.name,
      senderAvatar: currentUser.avatar,
      createdAt: new Date().toISOString()
    });
    setText('');
  };
}`}
